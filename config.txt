# =====================================================================
# JARVIS HUD - configuration
# ---------------------------------------------------------------------
# Format: key=value, one setting per line.
#
#   * UTF-8, no BOM. A byte order mark, if present, is ignored.
#   * Lines beginning with "#" are comments and are ignored.
#   * Whitespace around a key or a value is trimmed.
#   * Unknown keys are ignored, so it is safe to leave old settings
#     in place after an upgrade.
#
# THIS FILE IS RE-READ ON EVERY LAUNCH. There is no need to restart
# anything: edit a value, save, and the next greeting will use it.
# The file is looked for next to JarvisHud.exe. If it is missing, the
# built-in defaults below are used instead.
# =====================================================================

# --- Appearance ----------------------------------------------------

# Name used for the {name} placeholder in greetings.txt.
name=sir

# Monospace font family for all text. Only monospace faces are used.
# Fallback chain if unavailable: Cascadia Mono, Consolas,
# Lucida Console, Courier New.
font=Cascadia Mono

# Logical pixel size of the greeting text, scaled by the display DPI.
font_size=17

# Logical pixel size of the status line, scaled by the display DPI.
status_font_size=11

# --- Geometry ------------------------------------------------------

# Corner the HUD docks to, inside the primary monitor's work area.
# One of: bottom-right, bottom-left, top-right, top-left, center.
position=top-left

# Distance in logical pixels from the edge of the work area.
margin=32

# Logical width of the HUD panel, scaled by the display DPI.
panel_width=520

# Logical height of the HUD panel, scaled by the display DPI.
panel_height=186

# --- Timing --------------------------------------------------------

# Fade-in duration, in milliseconds.
fade_in_ms=450

# Reactor spin-up and sweep-line duration, in milliseconds.
boot_ms=1200

# Milliseconds per typed character in the typewriter reveal.
type_ms=28

# How long the finished greeting stays fully visible, in milliseconds.
hold_ms=4200

# Fade-out duration, in milliseconds.
fade_out_ms=700

# Hard lifetime limit in milliseconds, measured from launch.
# 0 = no limit; the HUD closes itself after fade_in + boot + typing
# + hold + fade_out. Any value greater than 0 forces the HUD to
# dismiss after that many milliseconds regardless of the other values.
dismiss_ms=0

# --- Behaviour -----------------------------------------------------

# 1 = keep the HUD on screen as a small corner widget after the
# greeting has been shown. 0 = show the greeting once and exit.
persist=0

# 1 = show the current time on the status line. 0 = hide it.
show_clock=1

# Text shown on the status line, and the value of the {status}
# placeholder in greetings.txt.
status_text=SYSTEM ONLINE

# 1 = keep the HUD above other windows (WS_EX_TOPMOST).
# 0 = the default. The HUD is then coverable by any application you
# open, and it never steals focus or blocks mouse clicks.
always_on_top=0

# 1 = also write the chosen greeting to the Winlogon
# LegalNoticeText value, which is displayed by Windows before the
# logon screen. This requires administrator rights and only takes
# effect the next time the machine starts.
notice_text=0
